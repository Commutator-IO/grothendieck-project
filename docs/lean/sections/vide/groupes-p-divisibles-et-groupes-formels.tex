\domainevide{Les cotes de ce thème supposent les groupes de Barsotti--Tate et la théorie de Dieudonné, absents de mathlib.}{129}

\begin{encadre}
\begin{conjecture}[Grothendieck, page 95, annulation de $\mathrm{Ext}^2$\piste{129-ext2-vanishing-p-odd}]
Si $2$ est inversible, le faisceau $\mathrm{Ext}^2_{\mathbf Z}(G, M)$ est nul pour tout schéma en groupes commutatif fini localement libre $G$ et tout module quasi-cohérent $M$, ce qui simplifie la théorie des déformations des groupes de Barsotti--Tate pour $p \neq 2$. Mathlib n'a pas les Ext de schémas en groupes finis localement libres.
\end{conjecture}
\end{encadre}
