\begin{encadre}
\begin{definition*}[Grothendieck, page 1, idéalisateur de Takeuchi\piste{155-takeuchi-idealiser}]
Soient $A$ un algébroïde de Hopf sur une base non centrale $k'$, $J = \mathrm{Ker}(k' \otimes_k k' \to k')$ et $\mathcal J = J\cdot(A \mathbin{\hat\otimes_k} A)$, idéal à droite. L'anneau où la comultiplication prend ses valeurs est le quotient idéalisateur $\tilde{\mathcal J}/\mathcal J$, avec $\tilde{\mathcal J} = \{\lambda \mid \lambda\mathcal J \subset \mathcal J\}$, et la comultiplication est un homomorphisme d'anneaux $A \to \tilde{\mathcal J}/\mathcal J$.
\end{definition*}
\end{encadre}
