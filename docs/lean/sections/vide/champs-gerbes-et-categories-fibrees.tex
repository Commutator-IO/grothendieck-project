\domainevide{Les cotes de ce thème supposent les champs et les gerbes sur un site, que mathlib ne définit pas.}{62, 63, 103, 134-1, 134-2, 134-8}
\noindent Pistes de l'onglet Findings pour ces cotes, hors de portée de Lean pour l'instant :
\begin{itemize}\small\setlength{\itemsep}{0pt}
\item Cote~62 : sur une base où $6$ est inversible, les triplets $(X, X', s)$ formés d'une courbe de genre $0$, d'une trisection étale et d'une section disjointe ont trois descriptions équivalentes, dont un invariant $j$ qui détermine le triplet là où $j(j-1)$ est inversible. Mathlib n'a pas les courbes de genre $0$ sur une base ni leurs trisections étales. On se reportera aux notes de Grothendieck (\lot{62}{3}{41}).\piste{62-triples-rigidified-moduli}
\end{itemize}

\begin{encadre}
\begin{conjecture*}[Grothendieck, pages 14 et 15, Gr-champs épinglés et cohomologie relative\piste{103-gr-stacks-relative-cohomology-BG}]
Soient $G$ un Groupe sur un topos $X$, non nécessairement commutatif, et $N$ un $G$-Module. Les Gr-champs sur $X$ épinglés par $(G, N)$, qui sont pour l'essentiel des 2-gerbes sur le topos classifiant $B_G$ liées par $N$ et trivialisées sur $X$, correspondent au complexe tronqué $\tau_{\leq 2}(R\Gamma(B_G \bmod X, N)[1])$, cohomologie de $B_G$ relative à $q_G : X \to B_G$.
\end{conjecture*}
\end{encadre}
{\small Mathlib n'a pas les Gr-champs ni les topos classifiants. On se reportera aux notes de Grothendieck (\lot{103}{1}{14}).\par}

\begin{encadre}
\begin{conjecture*}[Grothendieck, pages 3 à 5, suite exacte des champs de Picard épinglés\piste{134-1-picard-sign-sequence}]
Pour des faisceaux abéliens $M$, $N$ sur un topos, il existe une suite exacte de faisceaux $0 \to \mathrm{Ext}^2(M, N) \to P(M, N) \to \mathrm{Hom}(M, {}_2N) \to 0$, où $P(M, N)$ est le faisceau des champs de Picard épinglés par $M$, $N$ à équivalence près et où la dernière flèche est l'invariant de symétrie de $L \otimes L$. La page la tire, à titre heuristique, d'un triangle distingué.
\end{conjecture*}
\end{encadre}
{\small Mathlib n'a pas les champs de Picard sur un topos. On se reportera aux notes de Grothendieck (\lot{134-1}{1}{3}).\par}

\begin{encadre}
\begin{conjecture*}[Grothendieck, page 5, réalisation de l'invariant de symétrie\piste{134-1-sign-realisation-universal-argument}]
Sur tout topos, tout homomorphisme $M \to {}_2N$ est l'invariant de symétrie d'un champ de Picard épinglé par $M$, $N$ : l'obstruction, a priori dans $\mathrm{Ext}^3(M, N)$, est tuée par un argument \q{universel}, que la page annonce \q{sauf erreur} sans le donner.
\end{conjecture*}
\end{encadre}
{\small Mathlib n'a pas les champs de Picard ni leurs obstructions dans $\mathrm{Ext}^3$. On se reportera aux notes de Grothendieck (\lot{134-1}{1}{5}).\par}
