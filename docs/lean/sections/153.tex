\section[Cote 153 : polynômes à valeurs entières, co-opérations, analyseurs]{Cote 153 : polynômes à valeurs entières, co-opérations, analyseurs\protect\verifie{Folder153}}\label{sec:153}

La cote 153, \emph{Analyseurs : notes manuscrites (s.d.)}, dix pages que l'inventaire date \q{[à partir de 1982]}, cherche la structure abstraite d'un anneau d'opérations muni d'une composition --- ce qu'on appelle aujourd'hui une pléthorie ---, et y arrive par les polynômes à valeurs entières. La page~3 fixe une inclusion d'anneaux commutatifs $k_0 \subset K_0$ et l'anneau $\Omega = \{F \in K_0[T] \mid F(\lambda) \in k_0 \ \forall \lambda \in k_0\}$ ; pour que les co-opérations $F(x + y) = \sum_i G_i(x) H_i(y)$ et $F(xy) = \sum_\alpha J_\alpha(x) K_\alpha(y)$ soient des éléments \q{bien déterminés} de $\Omega \otimes_{k_0} \Omega$, elle énonce un lemme, le mot \q{Lemme} lui-même étant lu sans certitude : le sous-ensemble des $F \in K_0[T] \otimes_{k_0} M$ tels que $F(x) \in M$ pour tout $x \in k_0$ \q{s'identifie à $\Omega \otimes_{k_0} M$} (\lot{153}{1}{3}). La lecture précise : coordonnée par coordonnée dans une base de $M$.

Les autres feuillets portent la suite. La page~2 écrit les polynômes binomiaux $F_n = \binom{T}{n} \in \mathbf Q[T]$, affirme que $\bigoplus_n \mathbf Z F_n$ est une $\mathbf Z$-algèbre, et écrit la co-addition sous la forme générale $F_m(P + Q) = \sum \gamma^m_{ij} F_i(P) F_j(Q)$, la co-multiplication restant inachevée (\lot{153}{1}{2}). La page~5 définit l'analyseur, puis les constantes, et écrit sans justification $\Omega = \Omega_0 \oplus \Omega^0$ (\lot{153}{1}{5}). La page~7 donne les exemples $\mathrm{Appl}(A, A)$, $k[T]$ et $k[[T]]^+$, et l'application $\Omega \to \mathrm{Appl}(\Omega_0, \Omega_0) = \Omega_{\Omega_0}$, avec le souhait \q{que $\Omega \to \Omega_{\Omega_0}$ soit injectif !} (\lot{153}{1}{7}). La page~9 donne l'exemple relatif et l'analyseur binomial de $\mathbf Z \subset \mathbf Q$ (\lot{153}{1}{9}).

\begin{definition}
Soit $K_0$ une $k_0$-algèbre commutative ; on note $\iota : k_0 \to K_0$ le morphisme structural. On pose $\Omega = \{F \in K_0[T] \mid F(\iota(\lambda)) \in \iota(k_0) \ \forall \lambda \in k_0\}$ ; c'est une sous-$k_0$-algèbre de $K_0[T]$, les valeurs d'une somme et d'un produit étant la somme et le produit des valeurs, et un polynôme constant $\iota(c)$ y appartenant. Pour un $k_0$-module $M$, $x \in k_0$ et $F \in K_0[T] \otimes_{k_0} M$, on note $F(x) \in K_0 \otimes_{k_0} M$ l'image de $F$ par $\mathrm{ev}_x \otimes \mathrm{id}_M$, où $\mathrm{ev}_x : K_0[T] \to K_0$ est l'évaluation en $\iota(x)$, et l'on dit que $F(x) \in M$ lorsque $F(x)$ est dans l'image de $M \to K_0 \otimes_{k_0} M$, $m \mapsto 1 \otimes m$. On note $\Omega_M$ l'ensemble des $F$ tels que $F(x) \in M$ pour tout $x \in k_0$. Enfin \q{$\Omega_M$ s'identifie à $\Omega \otimes_{k_0} M$} est lu : l'application canonique $\Omega \otimes_{k_0} M \to K_0[T] \otimes_{k_0} M$ est injective, et son image est $\Omega_M$.
\end{definition}

\begin{enonce}[Lemme, page 3]
Soit $M$ un $k_0$-module libre, de base $(e_i)_{i \in I}$, et écrivons tout $F \in K_0[T] \otimes_{k_0} M$ sous la forme $\sum_i F_i \otimes e_i$, les $F_i \in K_0[T]$ étant presque tous nuls.
\begin{enumerate}
\item $F \in \Omega_M$ si et seulement si $F_i \in \Omega$ pour tout $i$.
\item L'application canonique $\Omega \otimes_{k_0} M \to K_0[T] \otimes_{k_0} M$ est injective.
\item Son image est exactement $\Omega_M$.
\end{enumerate}
\end{enonce}

Dans ce qui suit, $N$ et $P$ désignent des $k_0$-modules et $f : N \to P$ une application $k_0$-linéaire. La base $(e_i)$ donne, pour tout $N$, un isomorphisme $c_N : N \otimes_{k_0} M \to N^{(I)}$ sur les familles presque nulles d'éléments de $N$, qui envoie $n \otimes m$ sur $(m_i\, n)_i$, où les $m_i$ sont les coordonnées de $m$ ; c'est l'isomorphisme $N \otimes_{k_0} k_0^{(I)} \simeq N^{(I)}$. On appelle $c_N(t)_i$ la $i$-ième coordonnée de $t$.

\begin{lemme}\label{lem:153-nat}
Les coordonnées sont naturelles : pour $t \in N \otimes_{k_0} M$, la $i$-ième coordonnée de $(f \otimes \mathrm{id}_M)(t)$ est $f(c_N(t)_i)$.
\end{lemme}

\begin{proof}
Les deux membres sont additifs en $t$, et il suffit de le vérifier sur un tenseur pur $n \otimes m$ : la $i$-ième coordonnée de $f(n) \otimes m$ est $m_i f(n) = f(m_i n)$.
\end{proof}

\begin{lemme}\label{lem:153-image}
Un élément $t$ de $P \otimes_{k_0} M$ est dans l'image de $f \otimes \mathrm{id}_M$ si et seulement si chacune de ses coordonnées est dans l'image de $f$.
\end{lemme}

\begin{proof}
Si $t = (f \otimes \mathrm{id})(s)$, ses coordonnées sont les $f(c_N(s)_i)$ par le lemme~\ref{lem:153-nat}. Réciproquement, supposons $c_P(t)_i = f(g_i)$ pour tout $i$, et prenons $g_i = 0$ hors du support, fini, de $c_P(t)$ ; la famille $(g_i)$ est presque nulle, et soit $s = c_N^{-1}\bigl((g_i)_i\bigr)$. Par le lemme~\ref{lem:153-nat}, les coordonnées de $(f \otimes \mathrm{id})(s)$ sont les $f(g_i)$, qui valent $c_P(t)_i$ sur le support et $0 = c_P(t)_i$ ailleurs ; $c_P$ étant injectif, $(f \otimes \mathrm{id})(s) = t$.
\end{proof}

\begin{lemme}\label{lem:153-un}
L'image de $M \to K_0 \otimes_{k_0} M$, $m \mapsto 1 \otimes m$, est l'image de $\iota \otimes \mathrm{id}_M : k_0 \otimes_{k_0} M \to K_0 \otimes_{k_0} M$.
\end{lemme}

\begin{proof}
La première application est la composée de l'isomorphisme $M \simeq k_0 \otimes_{k_0} M$, $m \mapsto 1 \otimes m$, et de $\iota \otimes \mathrm{id}_M$.
\end{proof}

\begin{proof}[Démonstration du lemme de la page 3]
(1) Soit $x \in k_0$. Par le lemme~\ref{lem:153-un}, $F(x) \in M$ signifie que $(\mathrm{ev}_x \otimes \mathrm{id})(F)$ est dans l'image de $\iota \otimes \mathrm{id}$, c'est-à-dire, par le lemme~\ref{lem:153-image}, que chacune de ses coordonnées est dans $\iota(k_0)$. Par le lemme~\ref{lem:153-nat}, sa $i$-ième coordonnée est $\mathrm{ev}_x(F_i) = F_i(\iota(x))$. Ainsi $F \in \Omega_M$ si et seulement si $F_i(\iota(x)) \in \iota(k_0)$ pour tout $x$ et tout $i$, c'est-à-dire si et seulement si $F_i \in \Omega$ pour tout $i$.

(2) Un module libre est plat, fait classique ; le produit tensoriel par $M$ conserve donc l'injectivité de l'inclusion $\Omega \hookrightarrow K_0[T]$.

(3) Par le lemme~\ref{lem:153-image} appliqué à l'inclusion $\Omega \hookrightarrow K_0[T]$, $F$ est dans l'image de $\Omega \otimes_{k_0} M$ si et seulement si chaque $F_i$ est dans $\Omega$, c'est-à-dire, par~(1), si et seulement si $F \in \Omega_M$.
\end{proof}

\begin{remarque}
L'hypothèse de la page que $k_0 \to K_0$ soit une inclusion n'intervient pas : le lemme vaut pour toute $k_0$-algèbre commutative $K_0$, $\Omega$ étant l'ensemble des polynômes qui envoient $k_0$ dans l'image de $k_0$.
\end{remarque}

\medskip
\noindent\textit{Stabilité par composition et co-opérations (page~3).} On garde les notations de la définition ; on écrit $K_0[X, Y]$ pour l'anneau des polynômes en deux variables sur $K_0$, et, pour $F \in K_0[T]$, $\Delta_a F = F(X + Y)$ et $\Delta_m F = F(XY)$. On note $\Phi : \Omega \otimes_{k_0} \Omega \to K_0[X, Y]$ l'application $k_0$-linéaire qui envoie $G \otimes H$ sur $G(X) H(Y)$.

\begin{enonce}[Page 3, stabilité]
Si $F, G \in \Omega$, alors $F \circ G \in \Omega$.
\end{enonce}

\begin{proof}
Pour $x \in k_0$, $G(\iota(x)) = \iota(y)$ pour un $y \in k_0$, et $(F \circ G)(\iota(x)) = F(\iota(y)) \in \iota(k_0)$.
\end{proof}

\begin{enonce}[Page 3, les co-opérations]
Supposons que $K_0$ soit un localisé $S^{-1} k_0$ de $k_0$, et que $\Omega$ soit un $k_0$-module libre.
\begin{enumerate}
\item L'application $\Phi$ est injective.
\item Son image est l'ensemble des $P \in K_0[X, Y]$ tels que $P(\iota(x), \iota(y)) \in \iota(k_0)$ pour tous $x, y \in k_0$.
\item Pour $F \in \Omega$, il existe un unique $t \in \Omega \otimes_{k_0} \Omega$ tel que $\Phi(t) = F(X + Y)$, et un unique $t'$ tel que $\Phi(t') = F(XY)$.
\item Si $\Phi\bigl(\sum_i G_i \otimes H_i\bigr) = F(X + Y)$, alors $F \circ (P + Q) = \sum_i (G_i \circ P)(H_i \circ Q)$ pour tous $P, Q \in K_0[T]$ ; si $\Phi\bigl(\sum_\alpha J_\alpha \otimes K_\alpha\bigr) = F(XY)$, alors $F \circ (PQ) = \sum_\alpha (J_\alpha \circ P)(K_\alpha \circ Q)$.
\end{enumerate}
\end{enonce}

Soit $(e_j)_{j \in I}$ une base du $k_0$-module $\Omega$. Deux lemmes traduisent l'hypothèse sur $K_0$ : toute famille finie d'éléments de $K_0$ admet un dénominateur commun $s \in S$, c'est-à-dire que les $\iota(s) a$ sont dans $\iota(k_0)$, et $\iota(s)$ est inversible dans $K_0$.

\begin{lemme}\label{lem:153-indep}
Les $e_j$, vus dans $K_0[T]$, sont linéairement indépendants sur $K_0$.
\end{lemme}

\begin{proof}
Soit $\sum_j a_j e_j = 0$, les $a_j \in K_0$ presque tous nuls, et $s \in S$ un dénominateur commun : $\iota(s) a_j = \iota(d_j)$, $d_j \in k_0$. Alors $\sum_j d_j e_j = \iota(s) \sum_j a_j e_j = 0$ dans $\Omega$, donc les $d_j$ sont nuls, puis $\iota(s) a_j = 0$ et $a_j = 0$, $\iota(s)$ étant inversible.
\end{proof}

\begin{lemme}\label{lem:153-engendre}
Les $e_j$ engendrent $K_0[T]$ comme $K_0$-module.
\end{lemme}

\begin{proof}
Soit $p \in K_0[T]$. On chasse les dénominateurs des coefficients : il existe $s \in S$ et $q \in k_0[T]$ tels que $\iota(s) p$ soit l'image de $q$. Cette image envoie $\iota(x)$ sur $\iota(q(x))$, elle est donc dans $\Omega$, combinaison $k_0$-linéaire des $e_j$, et $p = \iota(s)^{-1} \cdot \iota(s) p$ est combinaison $K_0$-linéaire des $e_j$.
\end{proof}

\begin{proof}[Démonstration des co-opérations]
(1) Tout $t \in \Omega \otimes_{k_0} \Omega$ s'écrit $\sum_j G_j \otimes e_j$, les $G_j \in \Omega$ étant les coordonnées de $t$ dans la base $(e_j)$ du second facteur, et $\Phi(t) = \sum_j G_j(X) e_j(Y)$. Le coefficient de $X^m$ dans $\Phi(t)$ est $\sum_j c_m(G_j)\, e_j(Y)$, où $c_m(G_j) \in K_0$ est le coefficient de $T^m$ dans $G_j$. Si $\Phi(t) = \Phi(t')$, le lemme~\ref{lem:153-indep} donne $c_m(G_j) = c_m(G'_j)$ pour tous $j$ et $m$, donc $G_j = G'_j$ et $t = t'$.

(2) Les valeurs de $G(X) H(Y)$ en $(\iota(x), \iota(y))$ sont les produits $G(\iota(x)) H(\iota(y)) \in \iota(k_0)$, et la condition passe aux sommes. Réciproquement, soit $P$ à valeurs dans $\iota(k_0)$. Le lemme~\ref{lem:153-engendre}, appliqué aux coefficients de chaque monôme $X^n$, écrit $P = \sum_j G_j(X)\, e_j(Y)$ avec des $G_j \in K_0[T]$. Pour $x \in k_0$, le polynôme $P(\iota(x), T) = \sum_j G_j(\iota(x))\, e_j$ est dans $\Omega$, donc égal à $\sum_j \iota(\beta_j) e_j$ pour des $\beta_j \in k_0$ ; le lemme~\ref{lem:153-indep} donne $G_j(\iota(x)) = \iota(\beta_j)$. Ainsi chaque $G_j$ est dans $\Omega$, et $P = \Phi\bigl(\sum_j G_j \otimes e_j\bigr)$.

(3) Pour $F \in \Omega$, $(\Delta_a F)(\iota(x), \iota(y)) = F(\iota(x + y))$ et $(\Delta_m F)(\iota(x), \iota(y)) = F(\iota(xy))$ sont dans $\iota(k_0)$ : l'existence vient de~(2), l'unicité de~(1).

(4) Soit $\mathrm{ev}_{P,Q} : K_0[X, Y] \to K_0[T]$ le morphisme d'anneaux qui fixe $K_0$ et envoie $X$ sur $P$, $Y$ sur $Q$. Il envoie $G(X) H(Y)$ sur $(G \circ P)(H \circ Q)$, $F(X + Y)$ sur $F \circ (P + Q)$ et $F(XY)$ sur $F \circ (PQ)$ ; on l'applique à l'égalité supposée.
\end{proof}

\begin{remarque}
L'hypothèse employée est que $K_0$ soit un localisé de $k_0$, l'exemple que la lecture donne de son hypothèse $K_0 \otimes_{k_0} K_0 = K_0$ ; la démonstration ne passe pas par l'identification $K_0[X] \otimes_{k_0} K_0[Y] = K_0[X, Y]$, l'application $\Phi$ arrivant directement dans $K_0[X, Y]$. La liberté de $\Omega$, que la marge de la page demande, sert aux deux lemmes et à l'écriture $t = \sum_j G_j \otimes e_j$. L'inclusion $k_0 \subset K_0$ n'est pas supposée ; elle découle d'ailleurs de la liberté de $\Omega$ dès que $K_0 \neq 0$, un $c \in k_0$ tel que $\iota(c) = 0$ annulant tous les $e_j$. Les règles de~(4) valent pour n'importe quel représentant, et ne demandent pas l'unicité ; celle-ci est ce qui fait des co-opérations des éléments \q{bien déterminés} de $\Omega \otimes_{k_0} \Omega$. La formule $F \circ \lambda T = \sum_\alpha J_\alpha(\lambda) K_\alpha$ de la page en est le cas $P = \lambda$, $Q = T$.
\end{remarque}

\noindent\textit{Pólya et Vandermonde (page~2).} Pour $k_0 = \mathbf Z \subset K_0 = \mathbf Q$, $\Omega = \mathrm{Int}(\mathbf Z)$ ; on pose $F_n = \binom{T}{n} = T(T-1)\cdots(T-n+1)/n!$, et l'on note $\Delta f(x) = f(x+1) - f(x)$ la différence finie.

\begin{enonce}[Page 2, théorème de Pólya]
Les $F_n$ sont dans $\mathrm{Int}(\mathbf Z)$ et linéairement indépendants sur $\mathbf Z$. Tout $P \in \mathrm{Int}(\mathbf Z)$ de degré $d$ s'écrit $P = \sum_{k \le d} c_k F_k$ avec $c_k = (\Delta^k P)(0) \in \mathbf Z$. Ainsi $(F_n)_{n \in \mathbf N}$ est une base du $\mathbf Z$-module $\mathrm{Int}(\mathbf Z)$.
\end{enonce}

\begin{proof}
La valeur de $F_n$ en un entier $m$ est le coefficient binomial généralisé $\binom{m}{n}$, entier. Si $\sum_{k \in s} g_k F_k = 0$, la fonction $f(m) = \sum_k g_k \binom{m}{k}$ est nulle sur $\mathbf N$ ; or $\Delta^i \binom{\cdot}{k}$ vaut en $0$ le symbole de Kronecker $\delta_{ik}$, donc $g_i = (\Delta^i f)(0) = 0$. Soit maintenant $P \in \mathrm{Int}(\mathbf Z)$ et $f(m) = P(m) \in \mathbf Z$ pour $m \in \mathbf N$. Les $c_k = (\Delta^k f)(0) = \sum_{j \le k} (-1)^{k-j} \binom{k}{j} f(j)$ sont entiers, et ce sont aussi les $(\Delta^k P)(0)$ ; ils sont nuls pour $k > d$, la $k$-ième différence d'un polynôme de degré $< k$ étant nulle. La formule de Newton $f(m) = \sum_{k \le m} \binom{m}{k} (\Delta^k f)(0)$ montre que $P$ et $\sum_{k \le d} c_k F_k$ coïncident sur $\mathbf N$, partie infinie de $\mathbf Q$ ; ils sont égaux.
\end{proof}

\begin{enonce}[Page 2, Vandermonde, et les co-opérations de $\mathrm{Int}(\mathbf Z)$]
Dans $\mathbf Q[X, Y]$, $F_m(X + Y) = \sum_{i + j = m} F_i(X) F_j(Y)$. Par suite l'unique élément de $\mathrm{Int}(\mathbf Z) \otimes_{\mathbf Z} \mathrm{Int}(\mathbf Z)$ qui représente $F_m(X + Y)$ est $\sum_{i+j=m} F_i \otimes F_j$ : les $\gamma^m_{ij}$ de la page valent $1$ si $i + j = m$ et $0$ sinon. De même $F_m(XY) = \sum_{i,j} \delta^m_{ij} F_i(X) F_j(Y)$ pour des entiers $\delta^m_{ij}$ uniquement déterminés.
\end{enonce}

\begin{proof}
L'identité est la formule de Chu--Vandermonde dans un anneau binomial, $\binom{r+s}{m} = \sum_{i+j=m} \binom{r}{i}\binom{s}{j}$, que mathlib démontre (\texttt{Ring.add\_choose\_eq}) et qu'on applique dans $\mathbf Q[X, Y]$ à $r = X$, $s = Y$ ; les morphismes d'anneaux commutent aux coefficients binomiaux généralisés, ce qui identifie $\binom{X}{i}$ à $F_i(X)$. Par le théorème de Pólya, $\mathrm{Int}(\mathbf Z)$ est libre, $\mathbf Q$ est le localisé de $\mathbf Z$ en les entiers non nuls, et l'énoncé des co-opérations s'applique : il donne l'unicité, et, pour $F_m(XY)$, l'existence ; les coefficients dans la base $(F_i \otimes F_j)$ sont les $\delta^m_{ij}$.
\end{proof}

\begin{remarque}
Les entiers $c^p_{m,n}$ et $d^p_{m,n}$ de la page s'en déduisent : $F_m F_n$ et $F_m \circ F_n$ sont dans $\mathrm{Int}(\mathbf Z)$, sous-anneau stable par composition, donc combinaisons entières des $F_p$, de degré au plus $m + n$ et $mn$ ; cette conséquence n'a pas reçu d'énoncé Lean séparé. La lecture attribue le théorème à G.~Pólya (1919) ; la démonstration suivie ici est celle des différences finies.
\end{remarque}

\noindent\textit{Constantes et décomposition (page~5).} Soit $\Omega$ un pseudo-anneau commutatif muni d'une loi $\circ$ associative, d'unité $T$, telle que $F \mapsto F \circ G$ soit un morphisme de pseudo-anneaux pour tout $G$ : ce sont les conditions~1) et~2). On note $\Omega_0 = \{\lambda \mid \lambda \circ F = \lambda \ \forall F\}$ et $\Omega^0 = \{F \mid F \circ 0 = 0\}$.

\begin{enonce}[Page 5, $\Omega = \Omega_0 \oplus \Omega^0$]
$\Omega_0$ est un sous-pseudo-anneau de $\Omega$ ; pour $F \in \Omega$ et $\lambda \in \Omega_0$, $F \circ \lambda \in \Omega_0$, et en particulier $F \circ 0 \in \Omega_0$ ; $\Omega^0$ est un sous-groupe et un idéal ; et $\Omega$ est somme directe des sous-groupes $\Omega_0$ et $\Omega^0$, par $F = F \circ 0 + (F - F \circ 0)$. L'application $F \mapsto (\lambda \mapsto F \circ \lambda)$ de $\Omega$ dans $\mathrm{Appl}(\Omega_0, \Omega_0)$ respecte la somme, le produit et la composition, envoie $T$ sur l'identité, et envoie une constante $\lambda$ sur l'application constante de valeur $\lambda$.
\end{enonce}

\begin{proof}
Comme $F \mapsto F \circ G$ est additif, $0 \circ G = 0$ et $(-F) \circ G = -(F \circ G)$ ; additivité et multiplicativité montrent que $\Omega_0$ est stable par somme, opposé et produit. Si $\lambda \in \Omega_0$, $(F \circ \lambda) \circ G = F \circ (\lambda \circ G) = F \circ \lambda$, et $0 \in \Omega_0$. Si $F \in \Omega^0$, $(GF) \circ 0 = (G \circ 0)(F \circ 0) = 0$. Une constante $\lambda$ de $\Omega^0$ vérifie $\lambda = \lambda \circ 0 = 0$ : l'intersection est nulle. Enfin $(F - F \circ 0) \circ 0 = F \circ 0 - (F \circ 0) \circ 0 = 0$, $F \circ 0$ étant une constante, ce qui donne la somme. Les propriétés de $F \mapsto (\lambda \mapsto F \circ \lambda)$ sont les conditions~2), l'associativité, $T \circ \lambda = \lambda$ et la définition des constantes.
\end{proof}

\begin{remarque}
Les conditions~3) n'interviennent pas, ni l'unité $T$ sauf pour l'image de $T$ : la décomposition n'utilise que l'associativité et l'additivité de $F \mapsto F \circ G$, la multiplicativité ne servant qu'à faire de $\Omega_0$ un sous-pseudo-anneau et de $\Omega^0$ un idéal. La somme directe est une somme de groupes abéliens, $\Omega_0$ n'étant pas un idéal en général.
\end{remarque}

\noindent\textit{Exemples et contre-exemples (page~7).}

\begin{enonce}[Page 7, l'exemple $\mathrm{Appl}(A, A)$]
Soit $A$ un pseudo-anneau commutatif et $\Omega_A = \mathrm{Appl}(A, A)$, avec les lois ponctuelles et la composition des applications. Il vérifie~1) et~2) ; ses constantes sont les applications constantes, et $(\Omega_A)_0 \simeq A$ comme pseudo-anneaux. Si $A$ est un anneau commutatif unitaire non nul et infini, $\Omega_A$ ne vérifie pas~3a) : l'indicatrice $\delta$ de $0$ n'admet aucune famille finie $(F'_i, F''_i)_{i < n}$ telle que $\delta \circ (G' + G'') = \sum_i (F'_i \circ G')(F''_i \circ G'')$ pour tous $G'$, $G''$.
\end{enonce}

\begin{proof}
Une constante $\lambda$ vérifie $\lambda = \lambda \circ c_0$, où $c_0$ est l'application constante de valeur $0$, donc $\lambda(x) = \lambda(0)$ pour tout $x$ ; l'isomorphisme est $\lambda \mapsto \lambda(0)$. Pour~3a), prenons $G'$ et $G''$ constantes de valeurs $x$ et $y$ : on obtiendrait $\delta(x + y) = \sum_{i < n} F'_i(x) F''_i(y)$ pour tous $x, y \in A$. Choisissons $n + 1$ éléments distincts $a_0, \dots, a_n$ de $A$ et posons $U_{ji} = F'_i(a_j)$, $V_{ik} = F''_i(-a_k)$. Alors $(UV)_{jk} = \delta(a_j - a_k)$ est la matrice identité d'ordre $n + 1$, de rang $n + 1$ sur l'anneau commutatif non nul $A$, alors que le rang de $UV$ est au plus celui de $U$, qui a $n$ colonnes.
\end{proof}

\begin{enonce}[Page 7, l'analyseur des polynômes et l'application $\Omega \to \Omega_{\Omega_0}$]
Soit $k$ un anneau commutatif et $\Omega = k[T]$ avec la composition des polynômes. Il vérifie~1) et~2), et ses constantes sont les polynômes constants. Si $k$ est un corps fini à $q$ éléments, $T^q - T$ est non nul et $(T^q - T) \circ \lambda = 0$ pour toute constante $\lambda$ : l'application $\Omega \to \mathrm{Appl}(\Omega_0, \Omega_0)$ n'est pas injective. Si $k$ est un anneau intègre infini, elle l'est. Pour $\mathrm{Int}(\mathbf Z)$, deux polynômes $F, G \in \mathbf Q[T]$ tels que $F \circ n = G \circ n$ pour tout $n \in \mathbf Z$ sont égaux.
\end{enonce}

\begin{proof}
Pour une constante $\lambda$, $\lambda = \lambda \circ 0$ est le polynôme constant $\lambda(0)$ ; réciproquement un polynôme constant est une constante. Sur un corps fini, $a^q = a$ pour tout $a$, d'où $(T^q - T) \circ a = a^q - a = 0$, et $T^q - T$ est de degré $q \geq 2$. Sur un anneau intègre infini, comme sur $\mathbf Q$ avec la partie infinie $\mathbf Z$, un polynôme est déterminé par ses valeurs, et $F \circ a$ est le polynôme constant $F(a)$.
\end{proof}

\begin{enonce}[{Page 7, $k[[T]]^+$ ne vérifie pas 3a)}]
Soit $k$ un anneau commutatif non nul et $F = \sum_{m \ge 1} T^m$. Les coefficients de $X^p Y^q$ dans $F(X + Y) = \sum_{m \ge 1} (X + Y)^m$, qui valent $\binom{p+q}{p}$ hors de $(p, q) = (0, 0)$ et $0$ en $(0, 0)$, ne sont ceux d'aucune somme finie $\sum_{i < n} a_i(X) b_i(Y)$ de séries formelles : il n'existe pas de suites $(a_{i,p})$, $(b_{i,q})$ telles que ce coefficient soit $\sum_{i < n} a_{i,p} b_{i,q}$ pour tous $p, q$.
\end{enonce}

\begin{proof}
La matrice de Pascal $P_N = \bigl(\binom{p+q}{p}\bigr)_{p, q < N}$ est de déterminant $1$ : elle vaut $L L^{\mathsf T}$ avec $L = \bigl(\binom{p}{j}\bigr)_{p, j < N}$, par l'identité $\binom{p+q}{p} = \sum_j \binom{p}{j}\binom{q}{j}$, conséquence de Vandermonde, et $L$ est triangulaire à diagonale unité. Si les coefficients étaient $\sum_{i < n} a_{i,p} b_{i,q}$, ceux de $P_{n+2}$ seraient $\sum_{i < n} a_{i,p} b_{i,q} + [p = 0][q = 0]$, et $P_{n+2}$ serait le produit d'une matrice à $n + 1$ colonnes par une matrice à $n + 1$ lignes, de rang au plus $n + 1$, alors qu'elle est inversible, de rang $n + 2$.
\end{proof}

\begin{remarque}
Le contre-exemple de~3a) pour $\mathrm{Appl}(A, A)$ est établi pour $A$ unitaire, commutatif, non nul et infini ; la page ne parle pas de~3), et la lecture dit \q{pour $A$ infini}. Pour $k[[T]]^+$, la matrice des coefficients de $F(X + Y)$ n'est pas exactement la matrice de Pascal, comme le dit la note de la lecture : elle en diffère par le coefficient $(0,0)$, nul puisque $F$ n'a pas de terme constant, ce qui coûte un rang de plus dans l'argument. Le déterminant de la matrice de Pascal valant $1$ sur $\mathbf Z$, le contre-exemple vaut en toute caractéristique. L'énoncé Lean porte sur les coefficients ; le passage aux séries, coefficient de $X^p Y^q$ d'un produit $a(X) b(Y)$ égal à $a_p b_q$, est celui de la note.
\end{remarque}
