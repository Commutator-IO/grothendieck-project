\section[Cote 19 : le théorème de comonadicité 1.12, la comonade sur une base produit le théorème de Gabriel--Popescu et les chaînes d'adjoints]{Cote 19 : le théorème de comonadicité 1.12, la comonade sur une base produit, le théorème de Gabriel--Popescu et les chaînes d'adjoints\protect\verifie{Folder19}}\label{sec:19}

La cote 19, \emph{Topos : notes manuscrites (s.d.)}, quatre-vingt-quinze pages que l'inventaire date \q{[à partir de 1958-1973]}, est un dossier de travail en neuf liasses. La première, « Tapis cartésiens » (pages~3 à~11), étudie la comonade d'une adjonction et ses coalgèbres. La page~4 énonce le \q{Théorème 1.12}, suivi d'un nom propre que la transcription lit \q{Beck} avec doute. Elle écrit \q{il faut et il suffit} pour~b), et porte en marge la restriction \q{il suffit de regarder les doubles flèches $u, v$ de $A$ telles que $\mathrm{Ker}(f(u), f(v))$ existe \quad (condition C)} (\lot{19}{1}{4}). La deuxième liasse, « Cohomologie relative », s'ouvre pages~14 et~15 sur la représentabilité. Pour une catégorie à générateur $U$, elle affirme une équivalence avec les modules sur $\mathrm{End}(U)$, \q{[détails laissés au lecteur]} (\lot{19}{1}{15}).

\noindent\emph{Conventions.} On note, comme la lecture, $f : A \to B$ l'adjoint à gauche de $g : B \to A$, $\eta : \mathrm{id}_A \to gf$ l'unité, $\varepsilon : fg \to \mathrm{id}_B$ la coünité, et $\varphi = fg$ la comonade, de comultiplication $\delta_X = f(\eta_{gX})$. Une $\varphi$-coalgèbre est un couple $(X, a)$, $a : X \to \varphi X$, tel que $\varepsilon_X \circ a = 1_X$ et $\delta_X \circ a = \varphi(a) \circ a$ ; un morphisme $(X, a) \to (X', a')$ est une flèche $t : X \to X'$ telle que $a' \circ t = \varphi(t) \circ a$. Le foncteur de comparaison $h : A \to \varphi\text{-}\mathbf{Coalg}$ envoie $Y$ sur $(fY, f(\eta_Y))$ et $\psi$ sur $f(\psi)$. Le \emph{noyau} $\mathrm{Ker}(u, v)$ d'une double flèche est son égalisateur, et $f$ \emph{y commute} s'il envoie un noyau de $(u, v)$ sur un noyau de $(fu, fv)$. On utilise les identités triangulaires $\varepsilon_{fY} \circ f(\eta_Y) = 1$ et $g(\varepsilon_X) \circ \eta_{gX} = 1$, et la bijection d'adjonction $t \mapsto t^\flat = g(t) \circ \eta_Y$, de $\operatorname{Hom}_B(fY, X)$ sur $\operatorname{Hom}_A(Y, gX)$, d'inverse $s \mapsto \varepsilon_X \circ f(s)$.

\begin{definition}
Le foncteur $f$ vérifie la \emph{condition~C} si, pour toute double flèche $(u, v)$ de $A$ telle que $\mathrm{Ker}(fu, fv)$ existe, $\mathrm{Ker}(u, v)$ existe et $f$ y commute. Un triplet $(e, s, t)$ formé de $e : E \to Y$, $s : Y \to E$ et $t : Z \to Y$ est un \emph{noyau scindé} de $u, v : Y \rightrightarrows Z$ si $ue = ve$, $se = 1_E$, $tu = 1_Y$ et $tv = es$. Une double flèche $(u, v)$ de $A$ est \emph{$f$-scindée} si $(fu, fv)$ admet un noyau scindé.
\end{definition}

\begin{enonce}[Théorème 1.12, page 4]
Soit $f \dashv g$ une adjonction, $\varphi = fg$, et $h$ le foncteur de comparaison.
\begin{enumerate}
\item[a)] Si les noyaux de doubles flèches existent dans $A$ et si $f$ y commute, alors $h$ est essentiellement surjectif.
\item[b)] $h$ est une équivalence si et seulement si $f$ est conservatif et si, pour toute double flèche $(u, v)$ de $A$ telle que $\mathrm{Ker}(fu, fv)$ existe, $\mathrm{Ker}(u, v)$ existe et $f$ y commute.
\end{enumerate}
La restriction portée en marge (il suffit de considérer les doubles flèches $(u, v)$ dont l'image par $f$ a un noyau) est la condition~C.
\end{enonce}

\begin{lemme}\label{lem:19-scinde}
Si $(e, s, t)$ est un noyau scindé de $(u, v)$, alors $e$ est un noyau de $(u, v)$, et l'image de $(e, s, t)$ par tout foncteur est un noyau scindé de l'image de $(u, v)$.
\end{lemme}

\begin{proof}
Soit $z : T \to Y$ avec $uz = vz$. Alors $z = tuz = tvz = e(sz)$, donc $z$ se factorise par $e$, et de façon unique, puisque $ez' = ez''$ entraîne $z' = sez' = sez'' = z''$. Les quatre conditions sont des égalités entre composés, qu'un foncteur conserve.
\end{proof}

Un noyau scindé se dessine ainsi, les flèches $s$ et $t$ allant en sens inverse :
\[
\begin{tikzcd}[column sep=huge]
E \arrow[r, "e"] & Y \arrow[l, bend left=40, "s"] \arrow[r, shift left, "u"] \arrow[r, shift right, "v"'] & Z \arrow[l, bend left=40, "t"]
\end{tikzcd}
\qquad ue = ve, \quad se = 1_E, \quad tu = 1_Y, \quad tv = es.
\]

\begin{lemme}\label{lem:19-paire}
Pour toute coalgèbre $(X, a)$, la double flèche $(\eta_{gX}, g(a)) : gX \rightrightarrows gfgX$ est $f$-scindée : $(a, \varepsilon_X, \varepsilon_{fgX})$ est un noyau scindé de $(f(\eta_{gX}), fg(a))$. Elle est de plus coréflexive : $g(\varepsilon_X)$ est une rétraction commune de $\eta_{gX}$ et de $g(a)$.
\end{lemme}

\begin{proof}
On a $f(\eta_{gX}) \circ a = \delta_X \circ a = \varphi(a) \circ a = fg(a) \circ a$ par coassociativité ; $\varepsilon_X \circ a = 1_X$ par l'axiome de coünité ; $\varepsilon_{fgX} \circ f(\eta_{gX}) = 1$ par l'identité triangulaire en $gX$ ; et $\varepsilon_{fgX} \circ fg(a) = a \circ \varepsilon_X$ par naturalité de $\varepsilon$. Enfin $g(\varepsilon_X) \circ g(a) = g(\varepsilon_X \circ a) = 1$ et $g(\varepsilon_X) \circ \eta_{gX} = 1$.
\end{proof}

\begin{lemme}\label{lem:19-comp}
Supposons que, pour toute coalgèbre $(X, a)$, la double flèche $(\eta_{gX}, g(a))$ ait un noyau $e_X : R(X, a) \to gX$ et que $f$ y commute.
\begin{enumerate}
\item Pour tout $Y \in A$, $t \mapsto t^\flat$ induit une bijection entre les morphismes de coalgèbres $hY \to (X, a)$ et les flèches $k : Y \to R(X, a)$, par $t^\flat = e_X \circ k$.
\item Pour toute coalgèbre $(X, a)$, la flèche $\theta = \varepsilon_X \circ f(e_X)$ est un isomorphisme de coalgèbres de $h(R(X, a))$ sur $(X, a)$ ; en particulier $h$ est essentiellement surjectif.
\item Si de plus $f$ est conservatif, $h$ est une équivalence.
\end{enumerate}
\end{lemme}

\begin{proof}
(1) Soit $t : fY \to X$. Les transposées des deux membres de $a \circ t = \varphi(t) \circ f(\eta_Y)$ sont $g(a) \circ t^\flat$ et, puisque $\varphi(t) \circ f(\eta_Y) = f(t^\flat)$, $gf(t^\flat) \circ \eta_Y = \eta_{gX} \circ t^\flat$ par naturalité de $\eta$. La transposition étant bijective, $t$ est un morphisme de coalgèbres si et seulement si $t^\flat$ égalise $(\eta_{gX}, g(a))$, c'est-à-dire se factorise, de façon unique, par $e_X$.

(2) Posons $E = R(X, a)$. Par~(1) appliqué à $k = 1_E$, la flèche $\theta$, de transposée $e_X$, est un morphisme de coalgèbres $hE \to (X, a)$, et $a \circ \theta = \varphi(\theta) \circ f(\eta_E) = f(\theta^\flat) = f(e_X)$. Or $a$ est un noyau de $(f(\eta_{gX}), fg(a))$ par les lemmes~\ref{lem:19-scinde} et~\ref{lem:19-paire}, et $f(e_X)$ en est un par hypothèse ; donc $\theta$ est un isomorphisme. Son inverse est un morphisme de coalgèbres :
\[
\varphi(\theta^{-1}) \circ a = \varphi(\theta^{-1}) \circ a \circ \theta \circ \theta^{-1} = \varphi(\theta^{-1}) \circ \varphi(\theta) \circ f(\eta_E) \circ \theta^{-1} = f(\eta_E) \circ \theta^{-1}.
\]

(3) Soit $Y' \in A$. La flèche $\eta_{Y'}$ égalise $(\eta_{gfY'}, gf(\eta_{Y'}))$, par naturalité de $\eta$ ; c'est la double flèche attachée à la coalgèbre $hY' = (fY', f(\eta_{Y'}))$, et l'on a donc $\eta_{Y'} = e_{hY'} \circ k_{Y'}$ pour une flèche $k_{Y'} : Y' \to R(hY')$. Appliquons $f$ : $f(\eta_{Y'})$ est un noyau de l'image de cette double flèche (lemmes~\ref{lem:19-scinde} et~\ref{lem:19-paire}, avec $a = f(\eta_{Y'})$), $f(e_{hY'})$ aussi par hypothèse, et $f(e_{hY'}) \circ f(k_{Y'}) = f(\eta_{Y'})$ ; donc $f(k_{Y'})$ est un isomorphisme, et $k_{Y'}$ aussi, $f$ étant conservatif. Pour $\psi : Y \to Y'$, la transposée de $h(\psi) = f(\psi)$ est $gf(\psi) \circ \eta_Y = \eta_{Y'} \circ \psi = e_{hY'} \circ (k_{Y'} \circ \psi)$ : dans la bijection~(1), $h(\psi)$ correspond à $k_{Y'} \circ \psi$. Comme $\psi \mapsto k_{Y'} \circ \psi$ est bijective, $h$ est pleinement fidèle ; il est essentiellement surjectif par~(2), c'est donc une équivalence.
\end{proof}

\begin{proof}[Démonstration de a) et de la suffisance de b)]
Si $f$ vérifie la condition~C, chaque double flèche $(\eta_{gX}, g(a))$ a un noyau auquel $f$ commute, puisque son image a un noyau (lemmes~\ref{lem:19-paire} et~\ref{lem:19-scinde}). Le lemme~\ref{lem:19-comp}\,(2) donne alors a) sous la forme de la marge, et la forme du texte s'en déduit, car si tous les noyaux existent dans $A$ et que $f$ y commute, la condition~C est vérifiée. Si de plus $f$ est conservatif, $h$ est une équivalence par le lemme~\ref{lem:19-comp}\,(3).
\end{proof}

\begin{proof}[La nécessité de b) est fausse]
Soit $B$ la catégorie des couples d'ensembles $(X, Y)$, et $A$ la sous-catégorie pleine des couples tels que $Y \neq \varnothing$ entraîne $X \neq \varnothing$ ; $f$ est l'inclusion. Posons $g(X, Y) = (X, Y)$ si $X \neq \varnothing$ et $g(X, Y) = (\varnothing, \varnothing)$ sinon ; $g(X, Y)$ est dans $A$. Pour $(X', Y') \in A$, une flèche $(p, q) : (X', Y') \to (X, Y)$ de $B$ se factorise de façon unique par $g(X, Y) \to (X, Y)$ : si $Y' \neq \varnothing$, alors $X' \neq \varnothing$, donc $X \neq \varnothing$ ; si $Y' = \varnothing$, $q$ est l'application vide. Ainsi $f \dashv g$, et $f$ est une inclusion coréflexive. Une telle inclusion est pleinement fidèle, donc conservative, et elle est comonadique, résultat classique : la comonade est idempotente, et ses coalgèbres sont, à isomorphisme près, les objets de $A$. Le foncteur $h$ est donc une équivalence.

Mais $f$ ne vérifie pas la condition~C. Soient $u, v : (\ast, \ast) \rightrightarrows (\{0, 1\}, \ast)$ les flèches de $A$ qui envoient le point de la première composante sur $0$ et sur $1$ et sont l'identité sur la seconde. Leur noyau dans $B$ existe, c'est $(\varnothing, \ast)$. Si $K \to (\ast, \ast)$ était un noyau de $(u, v)$ dans $A$ auquel $f$ commute, $K$ serait isomorphe à $(\varnothing, \ast)$, de seconde composante non vide ; étant dans $A$, il aurait un point $x$ dans sa première composante, et $u$ et $v$ l'enverraient sur le même élément, c'est-à-dire $0 = 1$ : c'est absurde. Le noyau de $(u, v)$ dans $A$ est $(\varnothing, \varnothing)$, et $f$ n'y commute pas.
\end{proof}

\begin{enonce}[Théorème de Beck, forme exacte]
Le foncteur $h$ est une équivalence si et seulement si $f$ est conservatif et si, pour toute double flèche $f$-scindée $(u, v)$ de $A$, $\mathrm{Ker}(u, v)$ existe et $f$ y commute.
\end{enonce}

\noindent L'équivalence est démontrée dans le fichier de la section (\texttt{beck\_iff}), à partir des deux implications que mathlib contient séparément.

\noindent\emph{Une base produit.} Les pages~7 à~11 reprennent la comonade d'une adjonction quand la base est un produit $B = \prod_{i \in I} B_i$ : $f$ est une famille $f_i : A \to B_i$, $g$ envoie $(X_i)$ sur $\prod_i g_i(X_i)$, et la page pose $\varphi_{ji} = f_j g_i$ en supposant \q{que les $f_{j}$ commutent aux produits de type $I$ (p. ex. $f_{j}$ exacts ; ou $I$ fini)} (\lot{19}{1}{7}). Les données de la comonade y deviennent (3.9) $\bar\alpha_k = \alpha_k(X_k) \circ \mathrm{pr}_k$ et $\bar\lambda_{kji} = \lambda_{kji}(X_i) \circ \mathrm{pr}_i$, avec $\lambda_{kji} : \varphi_{ki} \to \varphi_{kj}\varphi_{ji}$ venu de l'unité $\mathrm{id} \to g_j f_j$ ; la marge note que $\lambda_{iii}$ est $\lambda_i$, et le cas de deux facteurs, $g'$, $g''$ pleinement fidèles, ne laisse que $\varphi'$, $\varphi''$, $\lambda'$, $\lambda''$ (\lot{19}{1}{9}). On se donne ici des adjonctions $f_i \dashv g_i$, $f_i : A \to B_i$, et l'on suppose que $A$ a les produits indexés par $I$ ; on note $\eta_i$, $\varepsilon_i$ leurs unités et coünités, $\mathrm{pr}_i$ les projections de $\prod_i g_i(X_i)$, et $\lambda_{kji}(X) = f_k(\eta_{j, g_i X}) : f_k g_i X \to f_k g_j f_j g_i X$.

\begin{enonce}[Pages 7 et 8, la matrice de la comonade\piste{19-comonad-matrix-product-base}]
\begin{enumerate}
\item Le foncteur $f = (f_i) : A \to \prod_i B_i$ est adjoint à gauche de $g$, $g((X_i)) = \prod_i g_i(X_i)$ ; l'unité a pour composantes les $\eta_i$, et la coünité en $(X_i)$ a pour composante $j$ la flèche $\varepsilon_{j, X_j} \circ f_j(\mathrm{pr}_j)$.
\item Si $f_j$ commute aux produits indexés par $I$, la composante $j$ de $\varphi((X_i))$ est un produit des $\varphi_{ji}(X_i)$, de projections les $f_j(\mathrm{pr}_i)$.
\item Sans autre hypothèse, la coünité de $\varphi$ a pour composante $k$ la flèche $\varepsilon_{k, X_k} \circ f_k(\mathrm{pr}_k)$, et la composante $k$ de la comultiplication $\delta$ vérifie, pour tous $j$ et $i$,
\[
f_k g_j f_j(\mathrm{pr}_i) \circ f_k(\mathrm{pr}_j) \circ \delta_k = \lambda_{kji}(X_i) \circ f_k(\mathrm{pr}_i) :
\]
c'est (3.9). Si tous les $f_j$ commutent aux produits indexés par $I$, ces égalités caractérisent $\delta_k$.
\item Pour $i = j = k$, $\lambda_{iii}$ est la comultiplication de la comonade $\varphi_i = f_i g_i$.
\end{enumerate}
\end{enonce}

L'adjonction du point~1, et la comonade vue comme une matrice de foncteurs $\varphi_{ji} = f_j g_i : B_i \to B_j$ (ici pour $I = \{1, 2\}$) :
\[
\begin{tikzcd}[column sep=huge]
A \arrow[r, shift left=1.2ex, "{f = (f_i)}"] & \prod_i B_i \arrow[l, shift left=1.2ex, "{g = \prod_i g_i}"]
\end{tikzcd}
\qquad
\varphi \leftrightarrow \begin{pmatrix} f_1 g_1 & f_1 g_2 \\ f_2 g_1 & f_2 g_2 \end{pmatrix}.
\]

\begin{proof}
(1) La bijection $\operatorname{Hom}(fY, (X_i)) = \prod_i \operatorname{Hom}(f_i Y, X_i) \simeq \prod_i \operatorname{Hom}(Y, g_i X_i) \simeq \operatorname{Hom}(Y, \prod_i g_i X_i)$ est naturelle en $Y$ et en $(X_i)$, composante par composante ; l'unité est l'image de l'identité, c'est-à-dire la flèche de composantes $\eta_{i, Y}$, et la coünité l'image réciproque de l'identité, de composante $j$ la transposée de $\mathrm{pr}_j$, soit $\varepsilon_{j, X_j} \circ f_j(\mathrm{pr}_j)$.

(2) Le cône $(\mathrm{pr}_i)$ est un produit dans $A$, et $f_j$ envoie un produit sur un produit : le cône $(f_j(\mathrm{pr}_i))$ de sommet $f_j(\prod_i g_i X_i)$ est un produit de la famille $f_j g_i(X_i)$.

(3) La coünité de $\varphi$ est celle de l'adjonction, calculée en~(1). La composante $k$ de $\delta$ est $f_k(\eta_{gX})$, où $\eta$ est l'unité de~(1), dont la composante $j$ est $\eta_j$. Donc $\delta_k$ suivie de $f_k(\mathrm{pr}_j)$ puis de $f_k g_j f_j(\mathrm{pr}_i)$ vaut $f_k\bigl(g_j f_j(\mathrm{pr}_i) \circ \eta_{j, gX}\bigr) = f_k\bigl(\eta_{j, g_i X_i} \circ \mathrm{pr}_i\bigr)$, par naturalité de $\eta_j$, c'est-à-dire $\lambda_{kji}(X_i) \circ f_k(\mathrm{pr}_i)$. Si les $f_j$ commutent aux produits indexés par $I$, $f_k(\prod_j g_j f_j g X)$ est un produit de projections $f_k(\mathrm{pr}_j)$ ; $g_j$, adjoint à droite, commute aux produits, si bien que $f_k g_j f_j(\prod_i g_i X_i)$ est un produit de projections $f_k g_j f_j(\mathrm{pr}_i)$. Deux flèches qui ont les mêmes composées avec toutes ces projections sont égales.

(4) $\lambda_{iii}(X) = f_i(\eta_{i, g_i X})$ est, par définition, la comultiplication de $f_i g_i$.
\end{proof}

\begin{enonce}[Pages 7 à 9, les cas dégénérés et les deux facteurs\piste{19-comonad-matrix-product-base}]
Supposons les $g_i$ pleinement fidèles.
\begin{enumerate}
\item $\varepsilon_i : \varphi_{ii} \to \mathrm{id}_{B_i}$ est un isomorphisme ; $\lambda_{kki}$ et $\lambda_{kjj}$ sont des isomorphismes pour tous $i, j, k$.
\item Pour $i \neq j$, $\lambda_{iji} = \lambda' \circ \varepsilon_i$, où $\lambda' = \lambda_{iji} \circ \varepsilon_i^{-1} : \mathrm{id}_{B_i} \to \varphi_{ij}\varphi_{ji}$.
\item Pour deux facteurs, $B = B' \times B''$, $\varphi' = f' g''$, $\varphi'' = f'' g'$ : si $f'$ commute aux produits binaires, la composante $B'$ de $\varphi(X', X'')$ est un produit de $X'$ et de $\varphi'(X'')$, de projections $\varepsilon'_{X'} \circ f'(\mathrm{pr}')$ et $f'(\mathrm{pr}'')$ ; de même, si $f''$ y commute, la composante $B''$ est un produit de $\varphi''(X')$ et de $X''$.
\end{enumerate}
\end{enonce}

\begin{proof}
(1) Un adjoint à droite pleinement fidèle a une coünité inversible. L'identité triangulaire $\varepsilon_{k, f_k Y} \circ f_k(\eta_{k, Y}) = 1$ montre alors que $f_k(\eta_{k, Y})$ est l'inverse de $\varepsilon_{k, f_k Y}$ ; avec $Y = g_i X$, c'est $\lambda_{kki}(X)$. L'autre identité, $g_j(\varepsilon_{j, X}) \circ \eta_{j, g_j X} = 1$, montre que $\eta_{j, g_j X}$ est l'inverse de $g_j(\varepsilon_{j, X})$, et $\lambda_{kjj}(X) = f_k(\eta_{j, g_j X})$ est inversible.

(2) C'est la définition de $\lambda'$.

(3) Par~(2) de l'énoncé précédent, la composante $B'$ de $\varphi(X', X'')$ est un produit de $f'g'(X')$ et de $f'g''(X'')$ ; remplacer la première projection par sa composée avec l'isomorphisme $\varepsilon'_{X'}$ donne encore un produit, de facteurs $X'$ et $\varphi'(X'')$. De même pour $B''$.
\end{proof}

\begin{encadre}
\begin{contrexemple}[Page 7, « ou $I$ fini »\piste{19-comonad-matrix-product-base}]
Il existe des adjonctions $f_i \dashv g_i$, avec $I$ à deux éléments, pour lesquelles la composante $j$ de $\varphi((X_i))$ n'est isomorphe au produit des $\varphi_{ji}(X_i)$ pour aucun choix d'isomorphisme.
\end{contrexemple}
\end{encadre}

\begin{proof}
Prenons $I = \{0, 1\}$, $A = B_0 = B_1$ la catégorie des ensembles, $f_i(Y) = \{0,1\} \times Y$ et $g_i(X) = X^{\{0,1\}}$, l'adjonction étant la curryfication, et $X_0 = X_1 = \ast$. Alors $g_i(\ast) = \ast$, $\prod_i g_i(X_i) = \ast$ et la composante $j$ de $\varphi(X)$ est $\{0,1\} \times \ast$, à deux éléments, tandis que $\prod_i \varphi_{ji}(X_i) = (\{0,1\} \times \ast)^2$ en a quatre.
\end{proof}

\begin{remarque}
La page suppose que les $f_j$ commutent aux produits indexés par $I$  : c'est l'hypothèse qui sert, et elle suffit. Elle en donne deux exemples. Le premier est juste : un foncteur exact à gauche commute aux produits finis. Le second est faux : un adjoint à gauche ne commute pas en général aux produits finis, et l'exemple ci-dessus est un cas où la formule $\mathrm{pr}_j\,\varphi((X_i)) = \prod_i \varphi_{ji}(X_i)$ est fausse avec $I$ fini. La lecture reprend la parenthèse de la page (« ce qui est automatique si $I$ est fini, ou si les $f_j$ sont exacts »). Il faut l'exactitude, ou directement la commutation aux produits.
\end{remarque}

\begin{remarque}
Les formules (3.9) ne demandent aucune hypothèse : elles décrivent les composées de la coünité et de la comultiplication avec les projections. La commutation des $f_j$ aux produits sert seulement à en déduire que ces données forment une matrice, c'est-à-dire qu'elles déterminent $\varphi$ et $\delta$. Les cas \q{dégénérés} de la page~8, $i = j$ ou $j = k$, sont, pour des $g_i$ pleinement fidèles, ceux où $\lambda_{kji}$ est un isomorphisme ; il ne reste donc, pour deux facteurs, que $\lambda'$ et $\lambda''$. La formule $\varphi(X', X'') = (X' \times \varphi'(X''), \varphi''(X') \times X'')$ demande encore que $f'$ et $f''$ commutent aux produits binaires, condition que la page tire de son hypothèse générale.
\end{remarque}

\begin{enonce}[Pages 14 et 15 ; théorème de Gabriel--Popescu\lie{4}]
Soient $C$ une catégorie de Grothendieck, $U$ un générateur de $C$ et $R = \mathrm{End}_C(U)$. Le foncteur $\operatorname{Hom}_C(U, -) : C \to \mathbf{Mod}_R$, à valeurs dans les $R$-modules à droite, est pleinement fidèle, et son adjoint à gauche $- \otimes_R U$ est exact.
\end{enonce}

\begin{proof}
La fidélité de $\operatorname{Hom}_C(U, -)$ est la définition d'un générateur. La pleine fidélité et l'exactitude à gauche de $- \otimes_R U$ sont le théorème de Gabriel--Popescu, publié par Popescu et Gabriel en 1964 \cite{popescugabriel1964}, que la preuve vérifiée invoque ; l'exactitude à droite vient de ce que $- \otimes_R U$ est un adjoint à gauche.
\end{proof}

\begin{remarque}
Sous la forme de la marge, a) n'utilise que la condition~C, et non la conservativité ; la suffisance de b) est le théorème de comonadicité de Beck, dual du théorème de monadicité (Mac~Lane 1998, §~VI.7, théorème~1 \cite{maclane1998}).
\end{remarque}

\begin{remarque}
L'erreur est sur la page : elle affirme la nécessité de b) par \q{il faut et il suffit}, alors que l'exemple ci-dessus est une adjonction comonadique dont l'adjoint à gauche ne vérifie pas la condition~C. La condition~b) ne coïncide donc pas avec le théorème de Beck sous sa forme exacte, qui caractérise les adjonctions comonadiques par une condition nécessaire et suffisante.
\end{remarque}

Toute double flèche $f$-scindée est une double flèche dont l'image a un noyau (lemme~\ref{lem:19-scinde}), et la réciproque est fausse. La condition~C est donc plus forte que celle de Beck : suffisante, elle n'est pas nécessaire.
\end{remarque}

\begin{remarque}
Le théorème sert en théorie des topos sous sa forme comonadique. Pour un morphisme de topos $u : E \to F$ dont l'image inverse $u^*$ est conservative (un morphisme surjectif), $u^*$ est adjoint à gauche de $u_*$, conservatif et exact à gauche, donc commute à tous les noyaux ; le théorème~1.12, sous la forme de la marge, et le lemme~\ref{lem:161-2-grossier} identifient alors $F$ à la catégorie des coalgèbres de la comonade $u^* u_*$ sur $E$. C'est la descente le long d'un morphisme surjectif, dont la page~34 de la cote~161-2 note un critère (section~\ref{sec:161-2}). Cette application est de l'édition ; elle n'est pas formalisée, mathlib n'ayant pas les morphismes de topos.
\end{remarque}

\paragraph{Chaînes d'adjoints (pages 79 et 80).} La page~80 prolonge une chaîne d'adjoints vers le bas par les foncteurs $f_!, f^*, f_*, f^!, \ldots$, énonce une Proposition, puis pose une Question : \q{Peut-on trouver une suite infinie dans les deux directions ?} La phrase s'achève page~81 sur une ébauche illisible (\lot{19}{4}{80}). La clause \q{dont aucun n'est pl.\ fid.} est une insertion interlinéaire. Pour un foncteur $u$ entre petites catégories, $u^*$ désigne la restriction des préfaisceaux d'ensembles le long de $u$.

\begin{theoreme}[Grothendieck, page 80, Proposition\piste{19-adjoint-chains-length-n}]\label{thm:19-chaine}
Pour tout $n$, il existe une petite catégorie $C$ et $n$ foncteurs $\widehat C \to \widehat C$, chacun adjoint à droite du précédent, dont aucun n'est pleinement fidèle.
\end{theoreme}

\begin{theoreme}[Page 80, la Question, énoncé de l'édition\piste{19-adjoint-chains-length-n}]\label{thm:19-infinie}
Soit $C = \mathbb Z \sqcup \mathbb Z$, réunion disjointe de deux copies de l'ensemble ordonné $\mathbb Z$. Il existe des foncteurs $F_k : \widehat C \to \widehat C$, $k \in \mathbb Z$, avec $F_k \dashv F_{k+1}$ pour tout $k$, dont aucun n'est fidèle, donc aucun pleinement fidèle. La Question de la page a donc une réponse positive, pour des topos de préfaisceaux sur une petite catégorie.
\end{theoreme}

\begin{proof}
Sur $\mathbb Z$, posons $e_m(x) = 2x + m$ et $r_m(x) = \lfloor (x - m)/2 \rfloor$. On a $2x + m \leqslant y \iff x \leqslant \lfloor (y-m)/2 \rfloor$ et $\lfloor (x-m)/2 \rfloor \leqslant y \iff x \leqslant 2y + m + 1$, d'où la chaîne de correspondances de Galois $e_m \dashv r_m \dashv e_{m+1}$, infinie dans les deux sens. Sur $C$, soient $B_{2m} = e_m \sqcup r_m$ et $B_{2m+1} = r_m \sqcup e_{m+1}$ ; alors $B_k \dashv B_{k+1}$ pour tout $k$. Une adjonction $u \dashv v$ entre petites catégories donne $u^* \dashv v^*$ sur les préfaisceaux, et l'on pose $F_k = B_k^*$.

Chaque $B_k$ agit sur l'une des deux copies par un $e_m$, dont l'image évite une classe de parité. Soit $q$ un point de cette copie hors de l'image, et $p = q - 1$ : tout $B_k(c) \leqslant q$ vérifie $B_k(c) \leqslant p$, et $q \not\leqslant p$. Soient $h_q$ le préfaisceau représenté par $q$ et $G$ le préfaisceau $x \mapsto \{b \in \{0, 1\} : x \leqslant p \Rightarrow b = 0\}$, dont les restrictions remplacent $b$ par $0$ au-dessous de $p$. Les deux morphismes $h_q \to G$ de valeurs $0$ et $1$ en $q$ sont distincts, et leurs restrictions par $B_k$ coïncident, puisque ces restrictions ne voient que des points au-dessous de $p$. Donc $F_k$ n'est pas fidèle. La Proposition s'obtient en prenant $F_0, \ldots, F_{n-1}$ (Lean : \texttt{proposition\_page80}, \texttt{chain\_adj}, \texttt{chain\_not\_fullyFaithful}).
\end{proof}

Sur une copie de $\mathbb Z$, la chaîne $\cdots \dashv r_{-1} \dashv e_0 \dashv r_0 \dashv e_1 \dashv r_1 \dashv \cdots$ se lit sur la figure suivante : $e_0$ (en trait plein) envoie $x$ sur $2x$, $e_1$ (en tirets) sur $2x + 1$, et $r_0$ renvoie $2x$ et $2x + 1$ sur $x$. L'image de $e_0$ évite les impairs (cercles blancs) : c'est le point $q$ de la démonstration.

Sur $C = \mathbb Z \sqcup \mathbb Z$, les deux copies sont décalées d'un cran : $B_{2m} = e_m \sqcup r_m$ et $B_{2m+1} = r_m \sqcup e_{m+1}$, de sorte que chaque $B_k$ agit sur l'une des copies par un $e$.

\begin{remarque}
La réunion disjointe est nécessaire : sur une seule copie de $\mathbb Z$, un $e_m^*$ sur deux est pleinement fidèle, puisque $r_m e_m = \mathrm{id}$. Les deux copies décalées d'un cran font qu'aucun $F_k$ ne l'est. Une note de la lecture, à la page~81, tient une chaîne infinie dans les deux sens pour impossible \q{dans des situations très générales} ; l'exemple montre qu'elle existe entre topos de préfaisceaux. La suggestion de la page, un $g$ à la fois adjoint à gauche et à droite de $f$, n'est pas celle suivie ici.
\end{remarque}

\begin{encadre}
\begin{theoreme}[Grothendieck, pages 7 à 11, comonade sur une base produit\piste{19-comonad-matrix-product-base}]\label{thm:19-quadruplets}
On reprend les adjonctions $f_i \dashv g_i$, $f_i : A \to B_i$, $A$ ayant les produits indexés par $I$.
\begin{enumerate}
\item Sans autre hypothèse, la matrice $(\varphi_{ji}, \varepsilon_i, \lambda_{kji})$ vérifie les relations de coünité et de coassociativité :
\[
\varphi_{ki}(\varepsilon_i) \circ \lambda_{kii} = 1, \qquad \varepsilon_k \varphi_{ki} \circ \lambda_{kki} = 1, \qquad \varphi_{lk}(\lambda_{kji}) \circ \lambda_{lki} = \lambda_{lkj}\varphi_{ji} \circ \lambda_{lji}.
\]
\item Si les $f_j$ commutent aux produits indexés par $I$, une flèche $a : X \to \varphi X$, d'entrées $u_{ki} = f_k(\mathrm{pr}_i) \circ a_k : X_k \to \varphi_{ki}(X_i)$, est une coalgèbre si et seulement si $\varepsilon_k \circ u_{kk} = 1$ et $\lambda_{kji}(X_i) \circ u_{ki} = \varphi_{kj}(u_{ji}) \circ u_{kj}$ pour tous $k, j, i$ ; une flèche $t : X \to Y$ vers une coalgèbre $(Y, b)$ d'entrées $v_{ki}$ est un morphisme de coalgèbres si et seulement si $v_{ki} \circ t_k = \varphi_{ki}(t_i) \circ u_{ki}$ pour tous $k, i$.
\item Deux facteurs, $B = B' \times B''$, $g'$ et $g''$ pleinement fidèles, $f'$ et $f''$ commutant aux produits binaires : la catégorie des $\varphi$-coalgèbres est équivalente à celle des quadruplets $(X', X'', u', u'')$, $u' : X' \to \varphi'(X'')$, $u'' : X'' \to \varphi''(X')$, tels que
\[
\lambda'(X') = \varphi'(u'') \circ u', \qquad \lambda''(X'') = \varphi''(u') \circ u'',
\]
un morphisme de quadruplets étant un couple $(t', t'')$ qui commute à $u'$ et à $u''$. Si de plus $f = (f', f'')$ est comonadique, $A$ est équivalente à la catégorie des quadruplets, par $Y \mapsto (f'Y, f''Y, f'(\eta''_Y), f''(\eta'_Y))$.
\end{enumerate}
\end{theoreme}
\end{encadre}

\begin{proof}
(1) Avec $\lambda_{kji}(X) = f_k(\eta_{j, g_i X})$, la première relation est $f_k$ appliqué à l'identité triangulaire $g_i(\varepsilon_{i, X}) \circ \eta_{i, g_i X} = 1$, la deuxième est l'identité triangulaire $\varepsilon_{k, f_k Y} \circ f_k(\eta_{k, Y}) = 1$ en $Y = g_i X$. Les deux membres de la troisième sont $f_l$ appliqué à $g_k f_k(\eta_{j, g_i X}) \circ \eta_{k, g_i X}$ et à $\eta_{k, g_j f_j g_i X} \circ \eta_{j, g_i X}$, égaux par naturalité de $\eta_k$ (Lean : \texttt{lam\_counit\_right}, \texttt{lam\_counit\_left}, \texttt{lam\_coassoc}).

(2) La coünité de $\varphi$ a pour composante $k$ la flèche $\varepsilon_k \circ f_k(\mathrm{pr}_k)$ (énoncé des pages~7 et~8, point~3) : la condition de coünité, composante par composante, est $\varepsilon_k \circ u_{kk} = 1$, et ceci sans hypothèse. Pour la coassociativité, on compose les deux membres de $\delta \circ a = \varphi(a) \circ a$, en la composante $k$, avec les flèches $f_k g_j f_j(\mathrm{pr}_i) \circ f_k(\mathrm{pr}_j)$. Le membre de gauche donne $\lambda_{kji}(X_i) \circ u_{ki}$ par (3.9). Celui de droite donne $\varphi_{kj}(u_{ji}) \circ u_{kj}$, puisque $\varphi(a)$ suivi de $f_k(\mathrm{pr}_j)$ vaut $f_k g_j(a_j) \circ f_k(\mathrm{pr}_j)$. Ces flèches séparent les morphismes à valeurs dans $(\varphi^2 X)_k$ quand les $f_j$ commutent aux produits (même argument que pour (3.9)). Les morphismes se traitent de même avec les seules projections $f_k(\mathrm{pr}_i)$ (Lean : \texttt{counit\_iff}, \texttt{coassoc\_iff}, \texttt{hom\_iff}).

(3) Les $g_i$ étant pleinement fidèles, $\varepsilon_i$ est inversible et la coünité impose $u_{kk} = \varepsilon_k^{-1}$ : c'est en ce sens que \q{l'axiome de coünité est automatiquement satisfait}. Il reste les deux entrées croisées $u' = u_{12}$ et $u'' = u_{21}$. Des huit relations de coassociativité, celles où $j = i$ se réduisent à $\lambda_{kii} = \varphi_{ki}(\varepsilon_i^{-1})$, conséquence de (1) ; celles où $k = j \neq i$ à $\lambda_{kki} = \varepsilon_k^{-1}\varphi_{ki}$ et à la naturalité de $\varepsilon_k^{-1}$ ; les deux qui restent, $k = i \neq j$, sont exactement les deux triangles, puisque $\lambda' = \lambda_{iji} \circ \varepsilon_i^{-1}$. Dans l'autre sens, un quadruplet donne une flèche $a$ dont la composante $k$ a pour entrées $u_{kk} = \varepsilon_k^{-1}$, $u'$, $u''$, ce qui la définit puisque $(\varphi X)_k$ est un produit des $\varphi_{ki}(X_i)$, et (2) montre que c'est une coalgèbre. Le foncteur des coalgèbres vers les quadruplets, qui retient les entrées croisées, est donc fidèle, plein (les conditions diagonales sur un morphisme sont la naturalité de $\varepsilon_k^{-1}$) et essentiellement surjectif. Si $f$ est comonadique, on le compose avec le foncteur de comparaison $h$ ; l'entrée $u'$ de $h(Y)$ est $f'(\mathrm{pr}'') \circ f'(\eta_Y) = f'(\eta''_Y)$ (Lean : \texttt{versQuadruplets\_isEquivalence}, \texttt{quadruplets}, \texttt{quadruplet\_u'}, \texttt{quadruplet\_u''}).
\end{proof}

\begin{encadre}
\begin{theoreme}[Grothendieck, pages 10 et 11, le cas dégénéré\piste{19-comonad-matrix-product-base}]\label{thm:19-degenere}
On se place dans la situation du point~(3) du théorème~\ref{thm:19-quadruplets}, $f = (f', f'')$ comonadique, et l'on note $\Phi : A \to \mathrm{Quad}$, $Y \mapsto (f'Y, f''Y, f'(\eta''_Y), f''(\eta'_Y))$, l'équivalence qu'il fournit.
\begin{enumerate}
\item Si $\varphi'\varphi''$ et $\varphi''\varphi'$ sont à valeurs finales, tout $(X', X'', u', u'')$ est un quadruplet : les deux triangles sont sans contenu.
\item Sans hypothèse de dégénérescence, un quadruplet $Q$ est dans l'image essentielle de $\Phi g'$ si et seulement si $u''$ est inversible, et dans celle de $\Phi g''$ si et seulement si $u'$ l'est ; de façon équivalente, $Y \in A$ est dans l'image essentielle de $g'$ si et seulement si $f''(\eta'_Y)$ est inversible. Plus généralement, pour $I$ quelconque, $g_i$ pleinement fidèle et $f = (f_j)$ conservatif, $Y$ est dans l'image essentielle de $g_i$ si et seulement si $f_j(\eta_{i, Y})$ est inversible pour tout $j \neq i$.
\item L'intersection des deux images essentielles est formée des quadruplets où $u'$ et $u''$ sont inversibles ; dans le cas dégénéré, $X'$ et $X''$ y sont finals.
\end{enumerate}
\end{theoreme}
\end{encadre}

\begin{proof}
(1) Les deux membres de chaque triangle sont des flèches vers un objet final (Lean : \texttt{Quadruplet.\allowbreak{}degenere}).

(2) Pour $g_i$ pleinement fidèle, $Y$ est dans l'image essentielle de $g_i$ si et seulement si $\eta_{i, Y}$ est inversible (mathlib : \texttt{isIso\_unit\_app\_iff\_mem\_essImage}). Or $f_i(\eta_{i, Y})$ l'est toujours, par l'identité triangulaire $\varepsilon_{i, f_i Y} \circ f_i(\eta_{i, Y}) = 1$ et l'inversibilité de $\varepsilon_i$ ; un $f$ conservatif, ce qu'est tout foncteur comonadique, rend donc $\eta_{i, Y}$ inversible dès que les $f_j(\eta_{i, Y})$, $j \neq i$, le sont (Lean : \texttt{essImage\_iff}, \texttt{reflectsIso\_of\_comonadic}). Pour deux facteurs, $f''(\eta'_Y)$ est l'entrée $u''$ de $\Phi(Y)$ (\texttt{essImage\_g'\_iff}, \texttt{essImage\_g''\_iff}). Enfin $\Phi$ est pleinement fidèle et essentiellement surjective, et un isomorphisme de quadruplets $(t', t'') : P \to Q$ donne $u''_Q \circ t'' = \varphi''(t') \circ u''_P$, avec $t'$ et $t''$ inversibles, ce qui transporte l'inversibilité (Lean : \texttt{Quadruplet.\allowbreak{}isIso\_u''\_of\_iso}, \texttt{essImage\_g'\_quadruplets}, \texttt{essImage\_g''\_quadruplets}).

(3) La première assertion est la conjonction des deux précédentes (\texttt{essImage\_inter}). Si $u'$ et $u''$ sont inversibles, $\varphi'(u'') \circ u' : X' \to \varphi'\varphi''(X')$ l'est, et le but est final ; de même pour $X''$ (\texttt{final\_of\_inter}).
\end{proof}

\begin{remarque}
Le critère de la page, \q{image essentielle de $g'$ : $u''$ est un isomorphisme}, ne dépend pas de la dégénérescence : il vaut dès que $g'$, $g''$ sont pleinement fidèles et $f$ comonadique. Restent non formalisées l'équivalence de l'intersection avec la sous-catégorie pleine de $B'$ où $\lambda'$ est inversible et les trois conditions équivalentes pour que cette intersection soit ponctuelle.
\end{remarque}
