% Une fenêtre fictive de #1 cm sur #2 cm, coin inférieur gauche à l'origine,
% avec une barre de titre de 0,55 cm qui porte #3.
\newcommand{\fenetre}[3]{%
  \fill[white] (0,0) rectangle (#1,#2);
  \begin{scope}
    \clip[rounded corners=3pt] (0,0) rectangle (#1,#2);
    \fill[black!9] (0,#2-0.55) rectangle (#1,#2);
  \end{scope}
  \draw[black!30] (0,#2-0.55) -- (#1,#2-0.55);
  \draw[black!40, rounded corners=3pt] (0,0) rectangle (#1,#2);
  \foreach \i in {0,1,2} \fill[black!25] (0.3+\i*0.28,#2-0.275) circle (0.07);
  \node[font=\sffamily\scriptsize, text=black!65] at (#1/2,#2-0.275) {#3};
}
