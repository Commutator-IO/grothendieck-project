\section*{1. Boîte (3), \emph{Géométrie élémentaire schématique} (581 feuillets) : la filiation avec EGA~V}
{\small\color{gray!60!black}Déjà vu le 23 octobre : 549 feuillets de texte, dix-huit sections du 19 août au 1\textsuperscript{er} octobre 1992, plus 32 feuillets de \q{Gribouillis}. Depuis, deux brouillons d'EGA~V conservés à l'IHÉS traitent des mêmes sujets : un tapuscrit \q{Formes lisses, et formes quadratiques lisses} (vers 1969) et des notes \q{Résultant et discriminant}. La question : le volume de 1992 reprend-il ces brouillons, ou repart-il de zéro ?}\par
\begin{faits}
\fait{Un discriminant \q{corrigé}, moitié du discriminant en rang impair ($\tilde d = d/2$), à valeurs dans $\det(E)^{\otimes 2}$}
\fait{Le théorème : $Q$ lisse $\Leftrightarrow$ $\tilde d(Q)$ inversible $\Leftrightarrow$ $Q$ localement standard (formes $X_1X_2 + \cdots$, $+X_{2m+1}^2$ en rang impair)}
\fait{Formes quadratiques lisses de rang $n$ $\leftrightarrow$ torseurs sous le groupe orthogonal $O(n)$ ; réduction à $\mu_2 = O(n)/SO(n)$}
\fait{La caractéristique 2 traitée à part : $x^2 + yz$ singularité ordinaire, \q{forme lisse} vs \q{non dégénérée}}
\fait{Le résultant, ou le discriminant d'une forme de degré $n > 2$}
\fait{Le Pfaffien (\q{discriminant = carré du Pfaffien} en rang pair, car.~2)}
\fait{Un renvoi à EGA~IV, à EGA~V, à SGA~3 (SGAD~XXIV) ou à un \q{plan}}
\fait{Les involutions projectives : leur place, et leur lien éventuel avec la trialité de (43)}
\end{faits}

\section*{2. Boîte (43), pochette 1, \emph{Notions de \q{géographie combinatoire}} (juin 1996) : la suite de 1986 ?}
{\small\color{gray!60!black}Déjà vu le 23 octobre : des ensembles \q{forestiers} finis dont les éléments jouent le rôle de \q{secteurs géographiques}. Depuis, la lecture de Montpellier relie trois inédits : les \emph{Teichmülleries} (144, cartes sur les surfaces et revêtements de $\mathbf P^1 \smallsetminus \{0,1,\infty\}$, 1982), la \emph{Structure à l'infini des $M_{g,\nu}$} (147, hiver 1982--1983, le principe des deux étages de la tour de Teichmüller un an avant l'\emph{Esquisse}) et \emph{Vers une géométrie des formes} (156, juin--juillet 1986, une topologie sans points, faite de morceaux et de leurs incidences, dont l'objet s'appelle une \q{contrée}). La question : la géographie de 1996 reprend-elle la géométrie des formes, ou les cartes de 1982 ?}\par
\begin{faits}
\fait{Le vocabulaire de 1986 : lieux, strates, multistrates, figures, \q{contrées}, \q{magasin}, \q{atelier}}
\fait{Les \q{représentations géographiques} de 146 (p.~127) : graphes dont les sommets portent un genre, comme ceux du bord des $M_{g,\nu}$}
\fait{Les trois relations de 1986 : être une face de, être à l'intérieur de, être disjoint de}
\fait{Pas de points : les secteurs sont-ils les données primitives, comme les lieux en 1986 ?}
\fait{Les \q{forêts} : arbres, réseaux ou ensembles ordonnés, comme les segments et les réseaux de 156-1 à 156-3}
\fait{Les cartes de 1982 : drapeaux, groupe cartographique ($\tau_0, \tau_1, \tau_2$), dessins, revêtements}
\fait{Un renvoi à \emph{Analysis situs}, à \q{GF}, à 1986, à la topologie modérée ou à l'\emph{Esquisse}}
\fait{Une récurrence sur la dimension (le programme annoncé en 156-1 : une géométrie des formes de dimension $n$)}
\end{faits}
