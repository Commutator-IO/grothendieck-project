\marginal{\emph{\uncertain{Sur un corps}}
\begin{itemize}
  \item cas elliptique : $\Delta\neq 0$
  \item cas multiplicatif : $\Delta=0$, $c_4\neq 0$ (\uncertain{équiv.} à $c_6\neq 0$)
