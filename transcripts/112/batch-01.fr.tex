\section*{Linéarisation}

\note{cette page ne porte que ce titre, à l'encre en haut à droite ; il est
répété au même endroit de la page suivante, dans une équerre tracée à la plume}

\page{2}

Soit $L \in \mathrm{Ob}\,\hat{A}_{\mathrm{ab}}$.

On dit que $L$ est \emph{lisse} si, pour tout $a \to b$ dans $A$,
\[
\int_{A_{a}} L_{a} \longrightarrow \int_{A_{b}} L_{b}
\]
est un \uncertain{quasi-isomorphisme}\note{relu sur le fac-similé à 600 dpi (2026-11-23) : le mot commence par un $q$ à longue hampe et se termine en « -iso » ; lecture probable, non vérifiée par un lecteur}, les deux membres étant respectivement égaux à
\[
\int_{A} L^{(a)} \quad\text{et}\quad \int_{A} L^{(b)},
\qquad\text{soit encore}\qquad
L^{(a)} * L^{A^{\circ}} \quad\text{et}\quad L^{(b)} * L^{A^{\circ}}.
\]

\note{les deux colonnes d'égalités sont écrites l'une sous l'autre sur la page,
reliées par des doubles barres ; on les met ici en ligne. Le mot qui qualifie la
flèche --- isomorphisme, quasi-isomorphisme --- est écrit très vite et n'est pas
assuré}

On dit que $L$ est \emph{négligeable} si $L^{b}$ est \ill

c'est-à-dire si
\[
\int_{A_{a}} L \;=\; \int_{A} L^{(a)} \longrightarrow \ill
\]

Soit donc $L^{b} = \bigl(a \mapsto L^{(a)} \otimes \ill\bigr)$ dans
$D_{\bullet}(\hat{B}_{\mathrm{ab}})$. C'est un complexe \ill\ dans
$\hat{B}_{\mathrm{ab}}$ qui est \ill\ \uncertain{localement constant}, \uncertain{i.e.} \ill\ à coefficients \uncertain{tordus}.\note{« loc.\ const. », abrégé sur la page, relu sur le fac-similé à 600 dpi (2026-11-23) ; le mot qui précède est abrégé et n'est pas lu}

Soit \struck{$W_{\uncertain{q}}$ \ill} $\mathrm{Lisse}_{\mathbb{Z}}(A)$ la sous-catégorie pleine de
$\hat{A}_{\mathrm{ab}}$ formée des objets lisses.

\begin{enumerate}
\item[(1)] \struck{Lisse} $W_{y}^{-1}\,\mathrm{Lisse}_{\mathbb{Z}}(A)
\longrightarrow D^{lc}(\hat{B}_{k})$ : est-ce une
\emph{équivalence des catégories} ?\note{lecture de Michel Hua sur le fac-similé : le but est $D^{lc}(\hat{B}_{k})$, et « équivalence des catégories »}

\note{sous la flèche, reliée au but par un trait vertical, une seconde ligne porte
$D^{lc}_{\bullet}(\hat{A}_{k})$ ; le long du trait : « $\wr$ par Quillen,
\uncertain{peut} \add{se} définir aussi comme $M_{\bullet} \mapsto M_{\bullet}^{b}$ ».
Le but $D^{lc}(\hat{B}_{k})$ est donc équivalent, « par Quillen », à
$D^{lc}_{\bullet}(\hat{A}_{k})$, et la flèche (1) peut se définir comme
$M_{\bullet} \mapsto M_{\bullet}^{b}$. L'indice de $W$ se lit $y$ ou $q$, avec un
« ? » au-dessous ; l'indice de $\hat{A}$, comme celui de $\hat{B}$, semble un $k$
écrit sur « ab ». En regard : « OK pour $A = \Delta$, mais \uncertain{aussi} ;
OK pour $A = \Delta_{/X}$, où $X \in \mathrm{Ob}\,\hat{\Delta}$ ? ». Relu sur le
fac-similé à 600 dpi (2026-11-23), non vérifié par un lecteur ; la première
lecture avait « une question sur la définition, où se lit encore « comme
$M_{\bullet} \mapsto M^{b}$ » ». Au bas de la même colonne, sous un nouveau
$\wr$, entre crochets : « $D_{\bullet}(\mathrm{Ab}) \cong \mathrm{Hot}_{ab}$ si
$A$ est asphérique »}

\item[(2)] Si $L, M \in \mathrm{Lisse}_{\mathbb{Z}}(A)$, a-t-on
\[
\mathrm{Ext}^{i}_{\mathbb{Z}_{A}}(L, M) \longrightarrow
\mathrm{Ext}^{i}_{\mathbb{Z}_{B}}(L^{b}, M^{b})
\]
pour \uncertain{$i \geq 0$} ?

\note{relu sur le fac-similé à 600 dpi (2026-11-23), non vérifié par un lecteur.
Après la flèche, « pour $i \geq 0$ » (le signe peut être $>$). Au-dessous, entre
crochets : « serait-ce vrai \uncertain{même} sans supposer $L$, $M$ lisses ? et
\uncertain{peut-on} \uncertain{remplacer} les $L$, $M$ par $L_{\bullet}$,
$M_{\bullet}$ ? ». À droite, entre crochets, sous un $\wr$ qui le rattache au
second membre : « $\mathrm{Ext}_{\mathbb{Z}}(\int_{A} L, \int_{A} M)$ si $A$
asph. ». Le crochet « $D_{\bullet}(\mathrm{Ab}) \cong \mathrm{Hot}_{ab}$ si $A$
est asphérique », que la première lecture rattachait à cette question, est au
bas de la colonne de (1), sous un $\wr$ qui le relie à
$D^{lc}_{\bullet}(\hat{A}_{k})$}

\item[(3)] Supposons $L$ lisse, \struck{de \ill{} $(L^{b}) = 0$ pour $i \leq N$.
On cherche un $L'$,} \add{et $N \in \mathbb{N}$. On cherche des $L' \in
\mathrm{Ob}\,\hat{A}_{\mathrm{ab}}$ et} $L \to L'$, tel que $L^{b} \to L'^{b}$
identifie $L'^{b}$ au \uncertain{tronqué} \ill{} de $L^{b}$ en dim $\leq N$.
\note{relu sur le fac-similé à 600 dpi (2026-11-23), non vérifié par un
lecteur ; la première lecture s'arrêtait à « Supposons $L$ lisse ». La fin de
la première ligne est barrée d'un long trait ; « et $N \in \mathbb{N}$. On
cherche des $L' \in \mathrm{Ob}\,\hat{A}_{\mathrm{ab}}$ et » est écrit au-dessus,
dans l'interligne. Un mot de petite taille, illisible, est ajouté au-dessus de
« de $L^{b}$ » ; le signe $\leq$ de « dim $\leq N$ » est repassé}
\end{enumerate}
